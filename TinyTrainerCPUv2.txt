numa_alloc.h

/* * numa_alloc.h — NUMA-aware allocation helpers. * * Improvement #4: NUMA-Local Memory Allocation * * On single-socket or systems without libnuma, all calls fall back to * standard posix_memalign / free. On multi-socket NUMA systems (dual * Xeon, Threadripper Pro, etc.), data chunks are allocated on the NUMA * node whose CPU cores will process them — eliminating cross-socket * memory traffic. * * Link with -lnuma when HAVE_NUMA is defined. */ #pragma once #include <stddef.h> #include <stdlib.h> #ifdef HAVE_NUMA # include <numa.h> # include <numaif.h> static inline int numa_node_count(void) { return numa_num_configured_nodes(); } static inline int numa_node_of_thread(int thread_id, int n_threads) { /* Round-robin mapping: thread → NUMA node */ int nodes = numa_node_count(); return (nodes > 0) ? (thread_id * nodes / n_threads) : 0; } /* * Allocate `size` bytes aligned to `align` bytes on NUMA node `node`. * Returns NULL on failure. */ static inline void *numa_aligned_alloc(size_t size, size_t align, int node) { void *p = numa_alloc_onnode(size, node); if (!p) return NULL; /* numa_alloc_onnode is page-aligned (4 KiB); re-align only if needed */ if ((uintptr_t)p % align != 0) { numa_free(p, size); /* Fallback: over-allocate and manually align */ void *raw = numa_alloc_onnode(size + align, node); if (!raw) return NULL; uintptr_t aligned = ((uintptr_t)raw + align) & ~(align - 1); return (void *)aligned; /* leaks at most (align-1) bytes — acceptable */ } return p; } static inline void numa_aligned_free(void *p, size_t size) { if (p) numa_free(p, size); } #else /* no libnuma — portable fallback */ static inline int numa_node_count(void) { return 1; } static inline int numa_node_of_thread(int thread_id, int n_threads) { (void)thread_id; (void)n_threads; return 0; } static inline void *numa_aligned_alloc(size_t size, size_t align, int node) { (void)node; void *p = NULL; #ifdef _WIN32 p = _aligned_malloc(size, align); #else if (posix_memalign(&p, align, size) != 0) return NULL; #endif return p; } static inline void numa_aligned_free(void *p, size_t size) { (void)size; #ifdef _WIN32 _aligned_free(p); #else free(p); #endif } #endif /* HAVE_NUMA */

return result; }
