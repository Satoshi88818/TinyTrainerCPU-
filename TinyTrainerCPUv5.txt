/* * sparse.h — CSR sparse dataset support. * * v5.0 additions: * IMP #4 — Feature Hashing via MurmurHash3. * sparse_ds_hashed_alloc() creates a hash-backed weight * table of fixed size TINY_HASH_SIZE. The training loop * replaces w[col_idx[k]] with w[murmur3_idx(col_idx[k])]. * Keeps the weight vector small enough to stay in L3 cache * regardless of the number of unique raw features. * * IMP #5 — Indirect prefetching in sparse_batch_gradient(). * Prefetches weight entries for indices 8 steps ahead in * the sparse row to hide random-access latency on CSR data. */ #pragma once #include <stddef.h> #include <stdlib.h> #include <string.h> #include "numa_alloc.h" #include "simd_compat.h" /* ----------------------------------------------------------------------- * * Compile-time tunables * * --------------------------------------------------------------------- */ #ifndef TINY_HASH_SIZE # define TINY_HASH_SIZE (1 << 20) /* 1M entries ≈ 4 MB, fits in L3 */ #endif #ifndef TINY_PREFETCH_DIST # define TINY_PREFETCH_DIST 8 /* steps ahead for indirect prefetch */ #endif /* ======================================================================= * * MurmurHash3 — 32-bit finalizer [IMP #4] * * * * Extremely fast integer hash. The output index is reduced modulo * * TINY_HASH_SIZE (power-of-2, so bitwise AND suffices). * * ===================================================================== */ static inline uint32_t murmur3_32(uint32_t x) { x ^= x >> 16; x *= 0x85ebca6bu; x ^= x >> 13; x *= 0xc2b2ae35u; x ^= x >> 16; return x; } /* Map a raw feature index to a hashed weight-table slot. */ static inline size_t tiny_hash_idx(int raw_col) { return (size_t)(murmur3_32((uint32_t)raw_col) & (uint32_t)(TINY_HASH_SIZE - 1)); } /* ======================================================================= * * CSR Sparse Dataset * * ===================================================================== */ typedef struct { float *values; /* [nnz] non-zero values */ int *col_idx; /* [nnz] raw column indices (before hashing) */ size_t *row_ptr; /* [n_samples+1] row start positions */ float *y; /* [n_samples] targets */ size_t n_samples; size_t n_features; size_t nnz; int use_hashing; /* 1 = apply feature hashing via tiny_hash_idx() */ } SparseDataset; static inline SparseDataset *sparse_ds_alloc(size_t n_samples, size_t n_features, size_t nnz) { SparseDataset *s = (SparseDataset *)calloc(1, sizeof(SparseDataset)); if (!s) return NULL; s->n_samples = n_samples; s->n_features = n_features; s->nnz = nnz; s->use_hashing = 0; s->values = (float *)numa_aligned_alloc(nnz * sizeof(float), 64, 0); s->col_idx = (int *)numa_aligned_alloc(nnz * sizeof(int), 64, 0); s->row_ptr = (size_t *)numa_aligned_alloc((n_samples+1)*sizeof(size_t), 64, 0); s->y = (float *)numa_aligned_alloc(n_samples * sizeof(float), 64, 0); if (!s->values || !s->col_idx || !s->row_ptr || !s->y) { numa_aligned_free(s->values, nnz * sizeof(float)); numa_aligned_free(s->col_idx, nnz * sizeof(int)); numa_aligned_free(s->row_ptr, (n_samples+1)*sizeof(size_t)); numa_aligned_free(s->y, n_samples * sizeof(float)); free(s); return NULL; } memset(s->row_ptr, 0, (n_samples + 1) * sizeof(size_t)); return s; } /* * sparse_ds_hashed_alloc [IMP #4] * Convenience constructor: identical to sparse_ds_alloc but sets * use_hashing=1. The caller trains against a weight vector of size * TINY_HASH_SIZE (allocated by model_alloc_hashed()) instead of * n_features, making the footprint constant. */ static inline SparseDataset *sparse_ds_hashed_alloc(size_t n_samples, size_t n_features, size_t nnz) { SparseDataset *s = sparse_ds_alloc(n_samples, n_features, nnz); if (s) s->use_hashing = 1; return s; } static inline void sparse_ds_free(SparseDataset *s) { if (!s) return; numa_aligned_free(s->values, s->nnz * sizeof(float)); numa_aligned_free(s->col_idx, s->nnz * sizeof(int)); numa_aligned_free(s->row_ptr, (s->n_samples + 1) * sizeof(size_t)); numa_aligned_free(s->y, s->n_samples * sizeof(float)); free(s); } /* ----------------------------------------------------------------------- * * Index selector: raw or hashed * * --------------------------------------------------------------------- */ static inline size_t sparse_widx(const SparseDataset *s, size_t k) { return s->use_hashing ? tiny_hash_idx(s->col_idx[k]) : (size_t)s->col_idx[k]; } /* ----------------------------------------------------------------------- * * sparse_dot — dot product for sample i * * --------------------------------------------------------------------- */ static inline double sparse_dot(const float * __restrict__ w, const SparseDataset *s, size_t i) { double acc = 0.0; for (size_t k = s->row_ptr[i]; k < s->row_ptr[i + 1]; ++k) acc += (double)w[sparse_widx(s, k)] * (double)s->values[k]; return acc; } /* ----------------------------------------------------------------------- * * sparse_batch_gradient [IMP #5 — Indirect Prefetching] * * * * For each non-zero in a sparse row, we issue a prefetch for the weight * * at index (k + TINY_PREFETCH_DIST). Because CSR non-zeros are random- * * access in the weight vector, without prefetching every weight access is * * a potential cache miss. Prefetching TINY_PREFETCH_DIST (default 8) * * steps ahead hides ~100–200 ns of DRAM latency. * * * * Prefetch hint: read (0), high temporal locality (3) — weights are * * re-read every batch so keeping them in L1 is worthwhile. * * --------------------------------------------------------------------- */ static inline void sparse_batch_gradient( const float * __restrict__ w, float b, const SparseDataset *s, const size_t *idx, size_t start, size_t end, double * __restrict__ grad_w, double * __restrict__ grad_b) { for (size_t bi = start; bi < end; ++bi) { size_t i = idx[bi]; size_t rend = s->row_ptr[i + 1]; double yhat = (double)b; /* --- Forward pass: indirect prefetch [IMP #5] --- */ for (size_t k = s->row_ptr[i]; k < rend; ++k) { /* Prefetch weight for the element TINY_PREFETCH_DIST ahead */ if (k + TINY_PREFETCH_DIST < rend) { size_t pre_widx = sparse_widx(s, k + TINY_PREFETCH_DIST); __builtin_prefetch(&w[pre_widx], 0, 3); } yhat += (double)w[sparse_widx(s, k)] * (double)s->values[k]; } double err = yhat - (double)s->y[i]; /* --- Backward pass: indirect prefetch [IMP #5] --- */ for (size_t k = s->row_ptr[i]; k < rend; ++k) { if (k + TINY_PREFETCH_DIST < rend) { size_t pre_widx = sparse_widx(s, k + TINY_PREFETCH_DIST); __builtin_prefetch(&grad_w[pre_widx], 1, 3); } grad_w[sparse_widx(s, k)] += err * (double)s->values[k]; } *grad_b += err; } } static inline float sparse_evaluate_mse(const float *w, float b, const SparseDataset *s) { double acc = 0.0; for (size_t i = 0; i < s->n_samples; ++i) { double err = (double)s->y[i] - (sparse_dot(w, s, i) + (double)b); acc += err * err; } return (float)(acc / (double)s->n_samples); } 

lookahead.h (new in v5.0)

/* * lookahead.h — Lookahead Optimizer (Zhang et al., 2019). [IMP #1] * * Lookahead wraps any inner optimizer (Adam, SGD+momentum) with a pair of * weight sets: * * "fast" weights — updated every batch by the inner optimizer (m->w). * "slow" weights — updated every k_steps batches via interpolation: * w_slow ← w_slow + α * (w_fast − w_slow) * * After the slow update, the fast weights are RESET to w_slow so the inner * optimizer explores from the new anchor. This acts as a global * synchronization that dramatically stabilises Hogwild! training and * reduces the need for tight omp critical sections on Adam updates. * * Typical hyperparameters: k_steps = 5, alpha = 0.5. */ #pragma once #include <stddef.h> #include <stdlib.h> #include <string.h> #include "simd_compat.h" #ifndef LOOKAHEAD_K # define LOOKAHEAD_K 5 /* inner steps between slow updates */ #endif #ifndef LOOKAHEAD_ALPHA # define LOOKAHEAD_ALPHA 0.5f /* interpolation coefficient */ #endif typedef struct { float *w_slow; /* slow weight copy, same size as model->w */ float b_slow; size_t padded_f; size_t step_count; /* inner optimizer steps since last slow update */ int k_steps; float alpha; } LookaheadState; static inline LookaheadState *lookahead_alloc(size_t padded_f, int k_steps, float alpha) { LookaheadState *la = (LookaheadState *)calloc(1, sizeof(LookaheadState)); if (!la) return NULL; la->w_slow = (float *)malloc(padded_f * sizeof(float)); if (!la->w_slow) { free(la); return NULL; } memset(la->w_slow, 0, padded_f * sizeof(float)); la->b_slow = 0.0f; la->padded_f = padded_f; la->step_count = 0; la->k_steps = k_steps; la->alpha = alpha; return la; } static inline void lookahead_free(LookaheadState *la) { if (la) { free(la->w_slow); free(la); } } /* * lookahead_sync — called once per inner optimizer step (batch). * Every k_steps calls it performs the slow-weights interpolation * and resets the fast weights. * * Returns 1 if a slow update was performed this call, 0 otherwise. */ static inline int lookahead_sync(LookaheadState * __restrict__ la, float * __restrict__ w_fast, float *b_fast, size_t n_features) { la->step_count++; if ((int)la->step_count < la->k_steps) return 0; la->step_count = 0; float alpha = la->alpha; /* w_slow += alpha * (w_fast - w_slow) → w_slow = (1-a)*w_slow + a*w_fast */ #if defined(__AVX2__) || defined(__AVX512F__) || \ defined(__ARM_NEON) || defined(__ARM_FEATURE_SVE) size_t vlen = VF_VLEN(); size_t simd = (n_features / vlen) * vlen; vecF va = VF_SET1(alpha); vecF v1a = VF_SET1(1.0f - alpha); for (size_t j = 0; j < simd; j += vlen) { vecF fast = VF_LOAD(w_fast + j); vecF slow = VF_LOAD(la->w_slow + j); vecF ns = VF_ADD(VF_MUL(v1a, slow), VF_MUL(va, fast)); VF_STORE(la->w_slow + j, ns); VF_STORE(w_fast + j, ns); /* reset fast to new slow */ } for (size_t j = simd; j < n_features; ++j) { float ns = (1.0f - alpha) * la->w_slow[j] + alpha * w_fast[j]; la->w_slow[j] = ns; w_fast[j] = ns; } #else for (size_t j = 0; j < n_features; ++j) { float ns = (1.0f - alpha) * la->w_slow[j] + alpha * w_fast[j]; la->w_slow[j] = ns; w_fast[j] = ns; } #endif la->b_slow = (1.0f - alpha) * la->b_slow + alpha * (*b_fast); *b_fast = la->b_slow; return 1; } 

swa.h (new in v5.0)

/* * swa.h — Stochastic Weight Averaging (SWA). [IMP #6] * * SWA maintains a running average of the model weights across the final * swa_start_frac (default 10%) of training epochs: * * w_swa = (1/n) * Σ w_i * * At the end of training, train() copies w_swa into model->w so the * returned model is the averaged weights. Architecture Decision Notes

Why simd_compat.h instead of per-file #ifdef __AVX2__? All SIMD intrinsic calls now route through VF_* macros. Adding a new architecture (e.g., RISC-V V extension) requires editing one file, not every kernel. The macro layer has zero runtime overhead — the compiler resolves it entirely at compile time.

Why Lookahead reduces omp critical pressure? In vanilla Hogwild! + Adam, every TINY_BLOCK_AGG batches each thread enters a critical section. The Lookahead slow-weights interpolation is a global sync of its own, so threads can use a longer k_steps interval (default 5 × TINY_BLOCK_AGG = 20 batches per global sync), dramatically reducing critical-section contention at the cost of one extra memcpy of the weight vector every k steps.

SWA and Early Stopping interaction: If early stopping fires before swa_start_epoch, no SWA snapshots are taken and the model is unmodified. This is the correct behavior: early stopping means the loss is already converged, so averaging is unnecessary.

Feature Hashing collision rate: With TINY_HASH_SIZE = 2^20 (≈1M) and a typical CTR dataset of 10M unique features, the expected collision rate is ~0.5% per weight slot. This is acceptable for most industrial sparse models and is the standard trade-off used in Vowpal Wabbit and libFFM.
