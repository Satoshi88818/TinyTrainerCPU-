/* * numa_alloc.h — NUMA-aware allocation helpers. * * v4.0 additions [OPT #18]: * Hyper-threading–aware thread→NUMA mapping. On modern CPUs, logical * core pairs (0,1), (2,3), … share one physical core, L1, and SIMD * execution units. The old linear mapping could place two Hogwild! * threads on the same physical core, causing severe L1 contention. * * New strategy: * 1. At init time, parse /proc/cpuinfo (Linux) or use * sched_getaffinity() to build a physical-core → logical-core map. * 2. numa_thread_to_pcpu() returns the n-th distinct physical core, * so Hogwild! threads are spread across physical cores first. * 3. NUMA node is derived from the chosen physical core via * /sys/devices/system/cpu/cpuN/node (Linux) or fallback. * * Falls back to the old linear behaviour on non-Linux or parse failure. */ #pragma once #include <stddef.h> #include <stdlib.h> #include <stdio.h> #include <string.h> /* ----------------------------------------------------------------------- * Physical-core topology (Linux only) * --------------------------------------------------------------------- */ #ifndef TINY_MAX_CPUS # define TINY_MAX_CPUS 1024 #endif typedef struct { int logical[TINY_MAX_CPUS]; /* logical → physical mapping */ int phys_to_first[TINY_MAX_CPUS]; /* physical → first logical core */ int n_logical; int n_physical; int valid; /* 1 if topology parsed successfully */ } CpuTopology; static inline CpuTopology *tiny_get_topology(void) { static CpuTopology topo; static int done = 0; if (done) return &topo; done = 1; memset(&topo, -1, sizeof(topo)); topo.n_logical = topo.n_physical = 0; topo.valid = 0; #ifdef __linux__ FILE *f = fopen("/proc/cpuinfo", "r"); if (!f) return &topo; char line[256]; int cur_logical = -1, cur_physical = -1; int seen[TINY_MAX_CPUS]; memset(seen, 0, sizeof(seen)); while (fgets(line, sizeof(line), f)) { int v; if (sscanf(line, "processor : %d", &v) == 1) { cur_logical = v; if (v >= 0 && v < TINY_MAX_CPUS) { if (v + 1 > topo.n_logical) topo.n_logical = v + 1; } } if (sscanf(line, "core id : %d", &v) == 1) { cur_physical = v; } /* End of a CPU block */ if (line[0] == '\n' && cur_logical >= 0 && cur_physical >= 0) { if (cur_logical < TINY_MAX_CPUS) { topo.logical[cur_logical] = cur_physical; if (!seen[cur_physical]) { seen[cur_physical] = 1; topo.phys_to_first[cur_physical] = cur_logical; topo.n_physical++; } } cur_logical = cur_physical = -1; } } fclose(f); topo.valid = (topo.n_physical > 0); #endif return &topo; } /* * Return the logical CPU index for the n-th physical core. * Falls back to n if topology is unavailable. */ static inline int numa_thread_to_pcpu(int thread_idx) { CpuTopology *t = tiny_get_topology(); if (!t->valid || t->n_physical == 0) return thread_idx; /* Wrap around if more threads than physical cores */ int phys_slot = thread_idx % t->n_physical; /* Walk phys_to_first to find the phys_slot-th physical core */ int count = 0; for (int i = 0; i < TINY_MAX_CPUS; ++i) { if (t->phys_to_first[i] >= 0) { if (count == phys_slot) return t->phys_to_first[i]; count++; } } return thread_idx; /* fallback */ } /* ----------------------------------------------------------------------- * NUMA node lookup for a logical CPU (Linux sysfs) * --------------------------------------------------------------------- */ static inline int numa_node_for_cpu(int logical_cpu) { #ifdef __linux__ char path[128]; snprintf(path, sizeof(path), "/sys/devices/system/cpu/cpu%d/node0", logical_cpu); /* node0 symlink exists only on the right node; scan node0..node7 */ for (int node = 0; node < 8; ++node) { char p2[128]; snprintf(p2, sizeof(p2), "/sys/devices/system/cpu/cpu%d/node%d", logical_cpu, node); FILE *f = fopen(p2, "r"); if (f) { fclose(f); return node; } } #else (void)logical_cpu; #endif return 0; } /* Convenience: NUMA node for thread_idx, HT-aware */ static inline int numa_node_of_thread(int thread_id, int n_threads) { (void)n_threads; int pcpu = numa_thread_to_pcpu(thread_id); return numa_node_for_cpu(pcpu); } static inline int numa_node_count(void) { #ifdef HAVE_NUMA return numa_num_configured_nodes(); #else CpuTopology *t = tiny_get_topology(); /* Count distinct NUMA nodes by scanning all physical CPUs */ int max_node = 0; for (int i = 0; i < t->n_logical && i < TINY_MAX_CPUS; ++i) { int nd = numa_node_for_cpu(i); if (nd > max_node) max_node = nd; } return max_node + 1; #endif } /* ----------------------------------------------------------------------- * Allocation helpers * --------------------------------------------------------------------- */ #ifdef HAVE_NUMA # include <numa.h> # include <numaif.h> static inline void *numa_aligned_alloc(size_t size, size_t align, int node) { void *p = numa_alloc_onnode(size, node); if (!p) return NULL; if ((uintptr_t)p % align != 0) { numa_free(p, size); void *raw = numa_alloc_onnode(size + align, node); if (!raw) return NULL; return (void *)(((uintptr_t)raw + align) & ~(align - 1)); } return p; } static inline void numa_aligned_free(void *p, size_t size) { if (p) numa_free(p, size); } #else static inline void *numa_aligned_alloc(size_t size, size_t align, int node) { (void)node; void *p = NULL; #ifdef _WIN32 p = _aligned_malloc(size, align); #else if (posix_memalign(&p, align, size) != 0) return NULL; #endif return p; } static inline void numa_aligned_free(void *p, size_t size) { (void)size; #ifdef _WIN32 _aligned_free(p); #else free(p); #endif } #endif /* HAVE_NUMA */

bi < end; ++bi) { size_t i = idx[bi]; double yhat = sparse_dot(w, s, i) + (double)b; double err = yhat - (double)s->y[i]; for (size_t k = s->row_ptr[i]; k < s->row_ptr[i + 1]; ++k) grad_w[s->col_idx[k]] += err * (double)s->values[k]; *grad_b += err; } } static inline float sparse_evaluate_mse(const float *w, float b, const SparseDataset *s) { double acc = 0.0; for (size_t i = 0; i < s->n_samples; ++i) { double err = (double)s->y[i] - (sparse_dot(w, s, i) + (double)b); acc += err * err; } return (float)(acc / (double)s->n_samples); }

(float)grad_w[j + k] : 0.0f; vecF g = VF_LOADU(g_buf); vecF m = VF_LOAD(s->m_w + j); vecF v = VF_LOAD(s->v_w + j); vecF wv = VF_LOAD(w + j); m = VF_ADD(VF_MUL(vb1, m), VF_MUL(v1mb1, g)); v = VF_ADD(VF_MUL(vb2, v), VF_MUL(v1mb2, VF_MUL(g, g))); wv = VF_SUB(wv, VF_MUL(vlrt, VF_DIV(m, VF_ADD(VF_SQRT(v), veps)))); VF_STORE(s->m_w + j, m); /* moments stay in cache (re-read next batch) */ VF_STORE(s->v_w + j, v); VF_STREAM(w + j, wv); /* OPT #19: weights bypass cache on write */ } /* Ensure non-temporal stores are visible before next read */ _mm_sfence(); for (size_t j = simd; j < n_features; ++j) { float g = (float)grad_w[j]; s->m_w[j] = beta1 * s->m_w[j] + (1.0f - beta1) * g; s->v_w[j] = beta2 * s->v_w[j] + (1.0f - beta2) * g * g; w[j] -= lr_t * s->m_w[j] / (sqrtf(s->v_w[j]) + eps); } #else for (size_t j = 0; j < n_features; ++j) { float g = (float)grad_w[j]; s->m_w[j] = beta1 * s->m_w[j] + (1.0f - beta1) * g; s->v_w[j] = beta2 * s->v_w[j] + (1.0f - beta2) * g * g; w[j] -= lr_t * s->m_w[j] / (sqrtf(s->v_w[j]) + eps); } #endif } /* ======================================================================= * Elastic Net (unchanged from God Tier v3.0) * ===================================================================== */ static void elastic_net_apply(float * __restrict__ w, size_t n_features, float l1_lambda, float l2_lambda, float lr) { float thresh = l1_lambda * lr; float scale = 1.0f / (1.0f + l2_lambda * lr); #if defined(__AVX512F__) size_t simd = (n_features / VEC_WIDTH) * VEC_WIDTH; vecF v_thresh = VF_SET1(thresh); vecF v_zero = VF_ZERO(); vecF v_scale = VF_SET1(scale); for (size_t j = 0; j < simd; j += VEC_WIDTH) { vecF wv = VF_LOAD(w + j); vecF abw = VF_ABS(wv); vecF st = VF_MUL( VF_DIV(wv, _mm512_max_ps(abw, VF_SET1(1e-30f))), VF_MAX(VF_SUB(abw, v_thresh), v_zero)); VF_STORE(w + j, VF_MUL(st, v_scale)); } for (size_t j = simd; j < n_features; ++j) { float sign = (w[j] >= 0.0f) ? 1.0f : -1.0f; w[j] = fmaxf(fabsf(w[j]) - thresh, 0.0f) * sign * scale; } #elif defined(__AVX2__) size_t simd = (n_features / VEC_WIDTH) * VEC_WIDTH; vecF v_thresh = VF_SET1(thresh); vecF v_zero = VF_ZERO(); vecF v_scale = VF_SET1(scale); __m256 v_sign_mask = _mm256_set1_ps(-0.0f); for (size_t j = 0; j < simd; j += VEC_WIDTH) { vecF wv = VF_LOAD(w + j); vecF sign = _mm256_and_ps(wv, v_sign_mask); vecF abw = VF_ABS(wv); vecF st = _mm256_or_ps(VF_MAX(VF_SUB(abw, v_thresh), v_zero), sign); VF_STORE(w + j, VF_MUL(st, v_scale)); } for (size_t j = simd; j < n_features; ++j) { float sign = (w[j] >= 0.0f) ? 1.0f : -1.0f; w[j] = fmaxf(fabsf(w[j]) - thresh, 0.0f) * sign * scale; } #else for (size_t j = 0; j < n_features; ++j) { float sign = (w[j] >= 0.0f) ? 1.0f : -1.0f; w[j] = fmaxf(fabsf(w[j]) - thresh, 0.0f) * sign * scale; } #endif } /* ======================================================================= * Shuffle & timer * ===================================================================== */ static uint64_t rng_state = 0x123456789ABCDEF0ULL; static inline uint64_t rng_next(void) { rng_state ^= rng_state << 13; rng_state ^= rng_state >> 7; rng_state ^= rng_state << 17; return rng_state; } static void shuffle_indices(size_t *idx, size_t n) { size_t block = TINY_SHUFFLE_BLOCK; size_t n_blocks = (n + block - 1) / block; for (size_t i = n_blocks - 1; i > 0; --i) { size_t j = (size_t)(rng_next() % (i + 1)); size_t ai = i * block, bi = j * block; size_t la = (ai + block <= n) ?
