/* * numa_alloc.h — NUMA-aware allocation helpers. * * Unchanged from Elite Edition except that numa_aligned_free now takes a * `size` parameter that is used correctly on all paths. * * Link with -lnuma when HAVE_NUMA is defined. */ #pragma once #include <stddef.h> #include <stdlib.h> #ifdef HAVE_NUMA # include <numa.h> # include <numaif.h> static inline int numa_node_count(void) { return numa_num_configured_nodes(); } static inline int numa_node_of_thread(int thread_id, int n_threads) { int nodes = numa_node_count(); return (nodes > 0) ? (thread_id * nodes / n_threads) : 0; } static inline void *numa_aligned_alloc(size_t size, size_t align, int node) { void *p = numa_alloc_onnode(size, node); if (!p) return NULL; if ((uintptr_t)p % align != 0) { numa_free(p, size); void *raw = numa_alloc_onnode(size + align, node); if (!raw) return NULL; return (void *)(((uintptr_t)raw + align) & ~(align - 1)); } return p; } static inline void numa_aligned_free(void *p, size_t size) { if (p) numa_free(p, size); } #else /* portable fallback */ static inline int numa_node_count(void) { return 1; } static inline int numa_node_of_thread(int t, int n) { (void)t; (void)n; return 0; } static inline void *numa_aligned_alloc(size_t size, size_t align, int node) { (void)node; void *p = NULL; #ifdef _WIN32 p = _aligned_malloc(size, align); #else if (posix_memalign(&p, align, size) != 0) return NULL; #endif return p; } static inline void numa_aligned_free(void *p, size_t size) { (void)size; #ifdef _WIN32 _aligned_free(p); #else free(p); #endif } #endif /* HAVE_NUMA */ 

sparse.h

/* * sparse.h — CSR (Compressed Sparse Row) dataset support. * * God Tier improvement: sparse gradient computation. * * For datasets where the average non-zero count per row (nnz/n) is much * smaller than n_features (e.g. text features, one-hot encoded categoricals), * CSR representation reduces gradient work from O(n·f) to O(nnz). * * This pairs naturally with Elastic Net (L1 sparsity in weights + sparse * inputs = cache-friendly updates on non-zero columns only). * * API mirrors the dense Dataset API so callers can swap them transparently * via the TrainConfig.use_sparse flag. */ #pragma once #include <stddef.h> #include <stdlib.h> #include <string.h> #include "numa_alloc.h" /* ----------------------------------------------------------------------- * * CSR dataset * * --------------------------------------------------------------------- */ typedef struct { float *values; /* [nnz] non-zero values */ int *col_idx; /* [nnz] column indices of each value */ size_t *row_ptr; /* [n_samples+1] start of each row in values/col_idx */ float *y; /* [n_samples] targets, dense */ size_t n_samples; size_t n_features; size_t nnz; /* total non-zeros */ } SparseDataset; /* Allocate a CSR dataset with known NNZ. */ static inline SparseDataset *sparse_ds_alloc(size_t n_samples, size_t n_features, size_t nnz) { SparseDataset *s = (SparseDataset *)calloc(1, sizeof(SparseDataset)); if (!s) return NULL; s->n_samples = n_samples; s->n_features = n_features; s->nnz = nnz; s->values = (float *)numa_aligned_alloc(nnz * sizeof(float), 64, 0); s->col_idx = (int *)numa_aligned_alloc(nnz * sizeof(int), 64, 0); s->row_ptr = (size_t *)numa_aligned_alloc((n_samples + 1) * sizeof(size_t), 64, 0); s->y = (float *)numa_aligned_alloc(n_samples * sizeof(float), 64, 0); if (!s->values || !s->col_idx || !s->row_ptr || !s->y) { numa_aligned_free(s->values, nnz * sizeof(float)); numa_aligned_free(s->col_idx, nnz * sizeof(int)); numa_aligned_free(s->row_ptr, (n_samples+1) * sizeof(size_t)); numa_aligned_free(s->y, n_samples * sizeof(float)); free(s); return NULL; } memset(s->row_ptr, 0, (n_samples + 1) * sizeof(size_t)); return s; } static inline void sparse_ds_free(SparseDataset *s) { if (!s) return; numa_aligned_free(s->values, s->nnz * sizeof(float)); numa_aligned_free(s->col_idx, s->nnz * sizeof(int)); numa_aligned_free(s->row_ptr, (s->n_samples + 1) * sizeof(size_t)); numa_aligned_free(s->y, s->n_samples * sizeof(float)); free(s); } /* * Sparse dot product: w · x_i (only sums over non-zero entries of row i). * Returns double for numerical safety. */ static inline double sparse_dot(const float * __restrict__ w, const SparseDataset *s, size_t i) { double acc = 0.0; size_t start = s->row_ptr[i]; size_t end = s->row_ptr[i + 1]; for (size_t k = start; k < end; ++k) acc += (double)w[s->col_idx[k]] * (double)s->values[k]; return acc; } /* * Compute sparse gradient for a single batch. * Only touches grad_w entries that correspond to non-zero columns — * all other entries remain untouched (zero), which is correct for SGD/Adam. */ static inline void sparse_batch_gradient( const float * __restrict__ w, float b, const SparseDataset *s, const size_t *idx, size_t start, size_t end, double * __restrict__ grad_w, double * __restrict__ grad_b) { for (size_t bi = start; bi < end; ++bi) { size_t i = idx[bi]; double yhat = sparse_dot(w, s, i) + (double)b; double err = yhat - (double)s->y[i]; /* accumulate sparse outer product */ size_t rs = s->row_ptr[i], re = s->row_ptr[i + 1]; for (size_t k = rs; k < re; ++k) grad_w[s->col_idx[k]] += err * (double)s->values[k]; *grad_b += err; } } /* MSE over the full sparse dataset */ static inline float sparse_evaluate_mse(const float *w, float b, const SparseDataset *s) { double acc = 0.0; for (size_t i = 0; i < s->n_samples; ++i) { double err = (double)s->y[i] - (sparse_dot(w, s, i) + (double)b); acc += err * err; } return (float)(acc / (double)s->n_samples); }

if (adam) adam_free(adam); if (vel_w) aligned_free(vel_w); free(idx); result.epochs_run = e; result.final_mse = cfg.use_sparse ? sparse_evaluate_mse(m->w, m->b, cfg.sparse_ds) : evaluate_mse(m, ds); result.train_time_sec = wall_time() - t_start; return result; }
