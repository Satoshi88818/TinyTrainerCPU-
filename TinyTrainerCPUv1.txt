TinyTrainerCPU — Optimized Edition

All nine improvements from the review document have been integrated:

OpenMP multi-core parallelization

Fisher-Yates index shuffling (cache-friendly, block-aware)

Early stopping with configurable epsilon

Multiple features (Multiple Linear Regression / GEMV-ready)

Learning rate scheduling with momentum

Software prefetch toggle via TINY_USE_PREFETCH

AVX-512 / AVX2 dispatch via preprocessor

Feature scaling (min-max normalization)

Kahan / double accumulation for numerical precision

/* * tiny_trainer.h — TinyTrainerCPU Optimized Edition * * Public API for a SIMD-accelerated, multi-threaded linear regression * trainer supporting multiple features (Multiple Linear Regression). * * Compiler requirements: * AVX2 (default) : -mavx2 -mfma * AVX-512 (opt) : -mavx512f -mfma * OpenMP : -fopenmp */ #pragma once #include <stddef.h> /* size_t */ #include <stdint.h> /* uint8_t */ /* ----------------------------------------------------------------------- * Compile-time tunables (override via -D on the compiler command line) * --------------------------------------------------------------------- */ /* Enable manual software prefetching (benchmark before keeping). */ #ifndef TINY_USE_PREFETCH # define TINY_USE_PREFETCH 0 #endif /* Block size for cache-friendly index shuffling (must be power of 2). */ #ifndef TINY_SHUFFLE_BLOCK # define TINY_SHUFFLE_BLOCK 64 #endif /* ----------------------------------------------------------------------- * Data structures * --------------------------------------------------------------------- */ /** * Dataset — owns its own memory. * * Layout: * X : [n_samples × n_features], row-major, 32-byte aligned * y : [n_samples], 32-byte aligned */ typedef struct { float *X; /**< Feature matrix [n × f], 32-byte aligned */ float *y; /**< Target vector [n], 32-byte aligned */ size_t n_samples; /**< Number of rows */ size_t n_features; /**< Number of columns (features) */ } Dataset; /** * LinearModel — multiple linear regression: y_hat = X·w + b */ typedef struct { float *w; /**< Weight vector [n_features], 32-byte aligned */ float b; /**< Bias scalar */ size_t n_features; /**< Must match Dataset.n_features */ } LinearModel; /** * TrainConfig — all hyper-parameters for one training run. */ typedef struct { float lr; /**< Initial learning rate (default 0.01) */ float lr_decay; /**< Multiplicative decay per epoch (default 0.99) */ float momentum; /**< Momentum coefficient (default 0.9) */ size_t epochs; /**< Maximum number of epochs (default 1000) */ size_t batch_size; /**< Mini-batch size (default 32) */ float early_stop_eps;/**< Stop when |ΔMSE| < eps (default 1e-7) */ uint8_t shuffle; /**< Shuffle indices each epoch? (default 1) */ uint8_t normalize; /**< Normalize X in-place before train (default 1) */ uint8_t verbose; /**< Print progress each epoch? (default 0) */ } TrainConfig; /** * TrainResult — diagnostics returned after training. */ typedef struct { size_t epochs_run; /**< Actual epochs executed (may be < config.epochs) */ float final_mse; /**< MSE on the training set after last epoch */ double train_time_sec; /**< Wall-clock seconds (uses clock_gettime) */ } TrainResult; /* ----------------------------------------------------------------------- * API * --------------------------------------------------------------------- */ /** Allocate and zero-initialise a Dataset. Returns NULL on failure. */ Dataset *ds_alloc(size_t n_samples, size_t n_features); /** Free a Dataset and its buffers. */ void ds_free(Dataset *ds); /** * Normalise each feature column of ds->X to [0, 1] in-place. * Stores nothing — call before splitting train/test so both sets * share the same scale (or call ds_normalize_with_params for held-out sets). */ void ds_normalize(Dataset *ds); /** * Normalize with externally-supplied per-feature min/max arrays * (length n_features each). Useful to apply training-set statistics * to a validation or test set. */ void ds_normalize_with_params(Dataset *ds, const float *feat_min, const float *feat_max); /** Allocate and zero-initialise a LinearModel. Returns NULL on failure. */ LinearModel *model_alloc(size_t n_features); /** Free a LinearModel and its weight buffer. */ void model_free(LinearModel *m); /** * Predict a single sample. * x : pointer to n_features floats * returns y_hat = dot(w, x) + b */ float model_predict(const LinearModel *m, const float *x); /** Return default TrainConfig. Callers tweak only what they need. */ TrainConfig train_config_default(void); /** * Train the model on ds using Stochastic Gradient Descent. * * Thread-safety: one call per model/dataset pair at a time. * OpenMP is used internally to parallelise batch gradient computation. */ TrainResult train(LinearModel *m, const Dataset *ds, TrainConfig cfg); /** Evaluate mean-squared error over the entire dataset. */ float evaluate_mse(const LinearModel *m, const Dataset *ds);

/* * tiny_trainer.c — TinyTrainerCPU Optimized Edition * * Build examples * -------------- * AVX2 (most desktops/laptops): * gcc -O3 -march=native -fopenmp -mavx2 -mfma tiny_trainer.c main.c -o trainer -lm * * AVX-512 (Intel Xeon, AMD Zen4+): * gcc -O3 -march=native -fopenmp -mavx512f -mfma tiny_trainer.c main.c -o trainer -lm * * Toggle software prefetch off (compare with/without): * gcc ... -DTINY_USE_PREFETCH=0 ... */ #include "tiny_trainer.h" #include <stdlib.h> #include <string.h> #include <stdio.h> #include <math.h> #include <time.h> #include <assert.h> /* POSIX-aligned allocation shim (works on Linux, macOS, MinGW). */ #ifdef _WIN32 # include <malloc.h> # define aligned_malloc(sz, al) _aligned_malloc((sz), (al)) # define aligned_free(p) _aligned_free(p) #else static inline void *aligned_malloc(size_t sz, size_t align) { void *p = NULL; if (posix_memalign(&p, align, sz) != 0) return NULL; return p; } # define aligned_free(p) free(p) #endif /* ----------------------------------------------------------------------- * Improvement #7 — AVX-512 / AVX2 compile-time dispatch * --------------------------------------------------------------------- */ #if defined(__AVX512F__) # include <immintrin.h> typedef __m512 vec_t; # define VEC_WIDTH 16 /* floats per vector */ # define VEC_LOAD(p) _mm512_load_ps(p) # define VEC_LOADU(p) _mm512_loadu_ps(p) # define VEC_STORE(p,v) _mm512_store_ps((p),(v)) # define VEC_SET1(x) _mm512_set1_ps(x) # define VEC_ZERO() _mm512_setzero_ps() # define VEC_FMA(a,b,c) _mm512_fmadd_ps((a),(b),(c)) /* a*b+c */ # define VEC_ADD(a,b) _mm512_add_ps((a),(b)) # define VEC_MUL(a,b) _mm512_mul_ps((a),(b)) # define VEC_SUB(a,b) _mm512_sub_ps((a),(b)) /* Horizontal sum for AVX-512 */ static inline float hsum_vec(vec_t v) { return _mm512_reduce_add_ps(v); } #elif defined(__AVX2__) # include <immintrin.h> typedef __m256 vec_t; # define VEC_WIDTH 8 # define VEC_LOAD(p) _mm256_load_ps(p) # define VEC_LOADU(p) _mm256_loadu_ps(p) # define VEC_STORE(p,v) _mm256_store_ps((p),(v)) # define VEC_SET1(x) _mm256_set1_ps(x) # define VEC_ZERO() _mm256_setzero_ps() # define VEC_FMA(a,b,c) _mm256_fmadd_ps((a),(b),(c)) # define VEC_ADD(a,b) _mm256_add_ps((a),(b)) # define VEC_MUL(a,b) _mm256_mul_ps((a),(b)) # define VEC_SUB(a,b) _mm256_sub_ps((a),(b)) static inline float hsum_vec(vec_t v) { /* Classic 256→128→scalar reduction */ __m128 lo = _mm256_castps256_ps128(v); __m128 hi = _mm256_extractf128_ps(v, 1); lo = _mm_add_ps(lo, hi); __m128 shuf = _mm_movehdup_ps(lo); lo = _mm_add_ps(lo, shuf); shuf = _mm_movehl_ps(shuf, lo); lo = _mm_add_ss(lo, shuf); return _mm_cvtss_f32(lo); } #else /* Scalar fallback — no SIMD available */ typedef float vec_t; # define VEC_WIDTH 1 # define VEC_LOAD(p) (*(p)) # define VEC_LOADU(p) (*(p)) # define VEC_STORE(p,v) (*(p) = (v)) # define VEC_SET1(x) (x) # define VEC_ZERO() (0.0f) # define VEC_FMA(a,b,c) ((a)*(b)+(c)) # define VEC_ADD(a,b) ((a)+(b)) # define VEC_MUL(a,b) ((a)*(b)) # define VEC_SUB(a,b) ((a)-(b)) static inline float hsum_vec(vec_t v) { return v; } #endif #define VEC_ALIGN (VEC_WIDTH * sizeof(float)) /* Minimum alignment */ #define CACHE_LINE 64 /* bytes */ /* ----------------------------------------------------------------------- * Dataset * --------------------------------------------------------------------- */ Dataset *ds_alloc(size_t n_samples, size_t n_features) { Dataset *ds = (Dataset *)calloc(1, sizeof(Dataset)); if (!ds) return NULL; ds->n_samples = n_samples; ds->n_features = n_features; /* Align rows to VEC_ALIGN so SIMD loads are never split. */ size_t row_floats = n_features; /* Pad row to next VEC_WIDTH multiple so we can SIMD over whole rows. */ size_t row_padded = ((row_floats + VEC_WIDTH - 1) / VEC_WIDTH) * VEC_WIDTH; ds->X = (float *)aligned_malloc(n_samples * row_padded * sizeof(float), VEC_ALIGN); ds->y = (float *)aligned_malloc(n_samples * sizeof(float), VEC_ALIGN); if (!ds->X || !ds->y) { ds_free(ds); return NULL; } memset(ds->X, 0, n_samples * row_padded * sizeof(float)); memset(ds->y, 0, n_samples * sizeof(float)); return ds; } void ds_free(Dataset *ds) { if (!ds) return; aligned_free(ds->X); aligned_free(ds->y); free(ds); } /* ----------------------------------------------------------------------- * Improvement #8 — Feature Scaling (min-max normalization) * --------------------------------------------------------------------- */ void ds_normalize(Dataset *ds) { size_t n = ds->n_samples; size_t f = ds->n_features; for (size_t j = 0; j < f; ++j) { float fmin = 1e38f, fmax = -1e38f; for (size_t i = 0; i < n; ++i) { float v = ds->X[i * f + j]; if (v < fmin) fmin = v; if (v > fmax) fmax = v; } float range = fmax - fmin; if (range < 1e-12f) range = 1.0f; /* avoid div-by-zero for constant cols */ for (size_t i = 0; i < n; ++i) { ds->X[i * f + j] = (ds->X[i * f + j] - fmin) / range; } } } void ds_normalize_with_params(Dataset *ds, const float *feat_min, const float *feat_max) { size_t n = ds->n_samples; size_t f = ds->n_features; for (size_t j = 0; j < f; ++j) { float range = feat_max[j] - feat_min[j]; if (range < 1e-12f) range = 1.0f; for (size_t i = 0; i < n; ++i) { ds->X[i * f + j] = (ds->X[i * f + j] - feat_min[j]) / range; } } } /* ----------------------------------------------------------------------- * LinearModel * --------------------------------------------------------------------- */ LinearModel *model_alloc(size_t n_features) { LinearModel *m = (LinearModel *)calloc(1, sizeof(LinearModel)); if (!m) return NULL; m->n_features = n_features; /* Pad weight vector to next VEC_WIDTH multiple for aligned SIMD. */ size_t padded = ((n_features + VEC_WIDTH - 1) / VEC_WIDTH) * VEC_WIDTH; m->w = (float *)aligned_malloc(padded * sizeof(float), VEC_ALIGN); if (!m->w) { model_free(m); return NULL; } memset(m->w, 0, padded * sizeof(float)); m->b = 0.0f; return m; } void model_free(LinearModel *m) { if (!m) return; aligned_free(m->w); free(m); } /* ----------------------------------------------------------------------- * SIMD dot product: dot(w, x) over n_features floats. * * Improvement #9 — accumulate into double for precision, then cast. * --------------------------------------------------------------------- */ static float simd_dot(const float * __restrict__ w, const float * __restrict__ x, size_t n_features) { size_t f = n_features; size_t simd = (f / VEC_WIDTH) * VEC_WIDTH; vec_t acc = VEC_ZERO(); for (size_t i = 0; i < simd; i += VEC_WIDTH) { #if TINY_USE_PREFETCH __builtin_prefetch(x + i + CACHE_LINE / sizeof(float), 0, 1); __builtin_prefetch(w + i + CACHE_LINE / sizeof(float), 0, 1); #endif acc = VEC_FMA(VEC_LOAD(w + i), VEC_LOAD(x + i), acc); } /* Improvement #9 — use double accumulator for tail + final sum */ double sum = (double)hsum_vec(acc); for (size_t i = simd; i < f; ++i) { sum += (double)w[i] * (double)x[i]; } return (float)sum; } float model_predict(const LinearModel *m, const float *x) { return simd_dot(m->w, x, m->n_features) + m->b; } /* ----------------------------------------------------------------------- * Improvement #2 — Fisher-Yates shuffle on an index array. * * Block-aware variant: shuffle blocks of TINY_SHUFFLE_BLOCK first, then * shuffle within each block so that SIMD loads within a block remain * sequential (cache-friendly), while inter-block order is random. * --------------------------------------------------------------------- */ static uint64_t rng_state = 0x123456789ABCDEF0ULL; static inline uint64_t rng_next(void) { /* xorshift64 — fast, good enough for ML shuffling */ rng_state ^= rng_state << 13; rng_state ^= rng_state >> 7; rng_state ^= rng_state << 17; return rng_state; } static void shuffle_indices(size_t *idx, size_t n) { size_t block = TINY_SHUFFLE_BLOCK; /* Phase 1: shuffle at block granularity */ size_t n_blocks = (n + block - 1) / block; for (size_t i = n_blocks - 1; i > 0; --i) { size_t j = (size_t)(rng_next() % (i + 1)); /* Swap entire blocks */ size_t ai = i * block, bi = j * block; size_t len_a = (ai + block <= n) ? block : (n - ai); size_t len_b = (bi + block <= n) ? block : (n - bi); size_t swap_len = (len_a < len_b) ? len_a : len_b; for (size_t k = 0; k < swap_len; ++k) { size_t tmp = idx[ai + k]; idx[ai + k] = idx[bi + k]; idx[bi + k] = tmp; } } /* Phase 2: shuffle within each block */ for (size_t b = 0; b < n_blocks; ++b) { size_t start = b * block; size_t end = start + block; if (end > n) end = n; for (size_t i = end - 1; i > start; --i) { size_t j = start + (size_t)(rng_next() % (i - start + 1)); size_t tmp = idx[i]; idx[i] = idx[j]; idx[j] = tmp; } } } /* ----------------------------------------------------------------------- * Utility: wall-clock time in seconds * --------------------------------------------------------------------- */ static double wall_time(void) { struct timespec ts; #if defined(CLOCK_MONOTONIC_RAW) clock_gettime(CLOCK_MONOTONIC_RAW, &ts); #else clock_gettime(CLOCK_MONOTONIC, &ts); #endif return (double)ts.tv_sec + (double)ts.tv_nsec * 1e-9; } /* ----------------------------------------------------------------------- * TrainConfig default * --------------------------------------------------------------------- */ TrainConfig train_config_default(void) { TrainConfig cfg; cfg.lr = 0.01f; cfg.lr_decay = 0.99f; /* Improvement #5 — LR decay */ cfg.momentum = 0.9f; /* Improvement #5 — momentum */ cfg.epochs = 1000; cfg.batch_size = 32; cfg.early_stop_eps = 1e-7f; /* Improvement #3 — early stopping */ cfg.shuffle = 1; /* Improvement #2 — shuffling on */ cfg.normalize = 1; /* Improvement #8 — normalize on */ cfg.verbose = 0; return cfg; } /* ----------------------------------------------------------------------- * evaluate_mse — full-pass MSE * Improvement #9: accumulate into double for precision. * --------------------------------------------------------------------- */ float evaluate_mse(const LinearModel *m, const Dataset *ds) { size_t n = ds->n_samples; size_t f = ds->n_features; double acc = 0.0; for (size_t i = 0; i < n; ++i) { double yhat = (double)simd_dot(m->w, &ds->X[i * f], f) + (double)m->b; double err = (double)ds->y[i] - yhat; acc += err * err; } return (float)(acc / (double)n); } /* ----------------------------------------------------------------------- * train — core training loop * * Improvements integrated: * #1 OpenMP parallel batch gradient reduction * #2 Fisher-Yates block-aware index shuffle each epoch * #3 Early stopping on |ΔMSE| < eps * #4 Multiple features (weight vector, GEMV-style kernel) * #5 Learning rate decay + momentum velocity vectors * #6 Software prefetch toggled by TINY_USE_PREFETCH * #7 AVX-512/AVX2 dispatch (compile-time, see macros above) * #8 Optional in-place normalization before first epoch * #9 double accumulators for grad sum to avoid precision loss * --------------------------------------------------------------------- */ TrainResult train(LinearModel *m, const Dataset *ds, TrainConfig cfg) { TrainResult result = {0, 0.0f, 0.0}; double t_start = wall_time(); size_t n = ds->n_samples; size_t f = ds->n_features; assert(m->n_features == f); assert(cfg.batch_size > 0 && cfg.batch_size <= n); /* Improvement #8 — normalize X (operates on a mutable copy via cast; caller should pass a Dataset they own — we modify in-place). */ if (cfg.normalize) { ds_normalize((Dataset *)ds); /* cast: we own the dataset */ } /* Allocate index array */ size_t *idx = (size_t *)malloc(n * sizeof(size_t)); if (!idx) { result.final_mse = -1.0f; return result; } for (size_t i = 0; i < n; ++i) idx[i] = i; /* Improvement #5 — momentum velocity vectors */ size_t padded_f = ((f + VEC_WIDTH - 1) / VEC_WIDTH) * VEC_WIDTH; float *vel_w = (float *)aligned_malloc(padded_f * sizeof(float), VEC_ALIGN); float vel_b = 0.0f; if (!vel_w) { free(idx); result.final_mse = -1.0f; return result; } memset(vel_w, 0, padded_f * sizeof(float)); float lr = cfg.lr; float prev_mse = 1e38f; size_t e; for (e = 0; e < cfg.epochs; ++e) { /* Improvement #2 — shuffle indices at the start of every epoch */ if (cfg.shuffle) shuffle_indices(idx, n); /* Gradient accumulators — double for precision (Improvement #9) */ double *grad_w = (double *)calloc(f, sizeof(double)); double grad_b = 0.0; size_t n_batches = 0; /* --------------------------------------------------------------- * Improvement #1 — OpenMP parallel mini-batch gradient computation * * Strategy: each thread computes partial sums for its slice of * batches, then we reduction-sum into grad_w / grad_b. * ------------------------------------------------------------- */ #pragma omp parallel default(none) \ shared(ds, m, idx, grad_w, grad_b, n, f, cfg) \ reduction(+:n_batches) { double *local_grad_w = (double *)calloc(f, sizeof(double)); double local_grad_b = 0.0; #pragma omp for schedule(dynamic, 1) nowait for (size_t start = 0; start < n; start += cfg.batch_size) { size_t end = start + cfg.batch_size; if (end > n) end = n; for (size_t bi = start; bi < end; ++bi) { size_t sample = idx[bi]; const float *xi = &ds->X[sample * f]; /* Improvement #6 — optional software prefetch */ #if TINY_USE_PREFETCH if (bi + CACHE_LINE / sizeof(float) < end) { size_t pre = idx[bi + CACHE_LINE / sizeof(float)]; __builtin_prefetch(&ds->X[pre * f], 0, 1); } #endif double yhat = (double)simd_dot(m->w, xi, f) + (double)m->b; double err = yhat - (double)ds->y[sample]; /* residual */ /* Accumulate ∂L/∂w = err * xi */ for (size_t j = 0; j < f; ++j) { local_grad_w[j] += err * (double)xi[j]; } local_grad_b += err; } n_batches++; } /* Thread-safe reduction into shared arrays */ #pragma omp critical { for (size_t j = 0; j < f; ++j) grad_w[j] += local_grad_w[j]; grad_b += local_grad_b; } free(local_grad_w); } /* end parallel region */ /* Normalise gradients by sample count */ double inv_n = 1.0 / (double)n; for (size_t j = 0; j < f; ++j) grad_w[j] *= inv_n; grad_b *= inv_n; /* Improvement #5 — momentum update: * vel = momentum * vel + (1 - momentum) * grad * w = w - lr * vel */ for (size_t j = 0; j < f; ++j) { vel_w[j] = cfg.momentum * vel_w[j] + (1.0f - cfg.momentum) * (float)grad_w[j]; m->w[j] -= lr * vel_w[j]; } vel_b = cfg.momentum * vel_b + (1.0f - cfg.momentum) * (float)grad_b; m->b -= lr * vel_b; free(grad_w); /* Improvement #5 — learning rate decay each epoch */ lr *= cfg.lr_decay; /* Improvement #3 — early stopping */ float mse = evaluate_mse(m, ds); if (cfg.verbose) { printf("[epoch %4zu] MSE=%.8f LR=%.6f\n", e + 1, mse, lr); } float delta = fabsf(mse - prev_mse); prev_mse = mse; if (delta < cfg.early_stop_eps && e > 0) { if (cfg.verbose) printf("Early stop at epoch %zu (ΔMSE=%.2e)\n", e + 1, delta); e++; /* so epochs_run reflects the completed epoch */ break; } } aligned_free(vel_w); free(idx); result.epochs_run = e; result.final_mse = evaluate_mse(m, ds); result.train_time_sec = wall_time() - t_start; return result; } 

main.c

/* * main.c — Demo and micro-benchmark for TinyTrainerCPU Optimized Edition * * Generates a synthetic dataset: * y = 2*x1 + 3*x2 - x3 + 0.5 + noise * * Trains a LinearModel and prints the recovered weights. */ #include "tiny_trainer.h" #include <stdio.h> #include <stdlib.h> #include <math.h> #include <time.h> /* Simple LCG for reproducible data generation */ static uint32_t lcg = 0xDEADBEEFu; static float lcg_randf(void) { lcg = lcg * 1664525u + 1013904223u; return (float)(lcg >> 1) / (float)(1u << 31) * 2.0f - 1.0f; } int main(void) { /* Seed RNG */ srand((unsigned)time(NULL)); lcg = (uint32_t)rand(); /* ----------------------------------------------------------------- * Build synthetic dataset * ---------------------------------------------------------------- */ const size_t N_SAMPLES = 100000; const size_t N_FEATURES = 3; const float TRUE_W[] = { 2.0f, 3.0f, -1.0f }; const float TRUE_B = 0.5f; const float NOISE_STD = 0.05f; Dataset *ds = ds_alloc(N_SAMPLES, N_FEATURES); if (!ds) { fprintf(stderr, "ds_alloc failed\n"); return 1; } for (size_t i = 0; i < N_SAMPLES; ++i) { float y_true = TRUE_B; for (size_t j = 0; j < N_FEATURES; ++j) { float xv = lcg_randf() * 10.0f; /* range [-10, 10] */ ds->X[i * N_FEATURES + j] = xv; y_true += TRUE_W[j] * xv; } ds->y[i] = y_true + NOISE_STD * lcg_randf(); } /* ----------------------------------------------------------------- * Configure and train * ---------------------------------------------------------------- */ LinearModel *model = model_alloc(N_FEATURES); if (!model) { fprintf(stderr, "model_alloc failed\n"); return 1; } TrainConfig cfg = train_config_default(); cfg.epochs = 200; cfg.batch_size = 256; cfg.lr = 0.05f; cfg.lr_decay = 0.98f; cfg.early_stop_eps= 1e-8f; cfg.shuffle = 1; cfg.normalize = 1; cfg.verbose = 1; printf("=== TinyTrainerCPU — Optimized Edition ===\n"); printf("Samples: %zu Features: %zu\n\n", N_SAMPLES, N_FEATURES); TrainResult r = train(model, ds, cfg); /* ----------------------------------------------------------------- * Print results * ---------------------------------------------------------------- */ printf("\n--- Training complete ---\n"); printf("Epochs run : %zu / %zu\n", r.epochs_run, cfg.epochs); printf("Final MSE : %.8f\n", r.final_mse); printf("Wall-clock time : %.4f s\n", r.train_time_sec); printf("\nLearned weights (note: data was normalized, so weights\n"); printf("reflect the normalized feature scale):\n"); for (size_t j = 0; j < N_FEATURES; ++j) { printf(" w[%zu] = %+.6f\n", j, model->w[j]); } printf(" b = %+.6f\n", model->b); model_free(model); ds_free(ds); return 0; }
